## Usage

const Example = () => {
  const [current, setCurrent] = useState(page);
  const onChange = (page: number, size: number) => {
    setCurrent(page);
    console.log(size);
  };
  return (
      <Pagination
        size={size}
        page={current}
        total={total}
        onChangePage={onChange}
      />
  );
};

export default Example;
